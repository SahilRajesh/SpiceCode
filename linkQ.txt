//Linked List implementation of a Queue.
#include <stdio.h>
#include<stdlib.h>

struct Node {
    int data;
    struct Node *next;
};
struct Node *front = NULL;
struct Node *rear = NULL;

int isEmpty(){
    return front ==NULL;
}

void enque(int val){
    struct Node *newNode = (struct Node*)malloc(sizeof(struct Node));
    if(newNode == NULL){
        printf("Queue overflow.");
        return;
    }
    newNode->data = val;
    newNode->next = NULL;
    if(rear == NULL){
        front = rear = newNode;
    }else{
        rear->next = newNode;
        rear = newNode;
    }
    printf("%d added onto queue\n",val);
}

void deque(){
    if(isEmpty()){
        printf("Queue underflow.");
        return;
    }
    struct Node *temp = front;
    front = front->next;
    if(front == NULL) {
        rear = NULL;
    }
    printf("Removed element: %d\n",temp->data);
    free(temp);
}

void display(){
    if(isEmpty()){
        printf("Queue empty.");
        return;
    }
    struct Node *temp = front;
    printf("Queue elements:\n");
    while(temp != NULL){
        printf("%d\t",temp->data);
        temp = temp->next;
    }
    printf("\n");
}


int main() {
    int val,choice=1;
    while(choice != 4){
        printf("\n--Linked list menu--\n");
        printf("1.Enqueue\n2.Dequeue\n3.Display\n4.Exit\n");
        printf("Enter your choice: ");
        scanf("%d",&choice);
        switch(choice){
            case 1:
                printf("Enter val to insert: ");
                scanf("%d",&val);
                enque(val);
                break;
            case 2:
                deque();
                break;
            case 3:
                display();
                break;
            case 4:
                printf("Exiting\n");
                break;
            default: 
                printf("Invalid choice.");
        }
    }
    return 0;
}
