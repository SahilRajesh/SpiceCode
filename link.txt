//Linked List implementation of stack.
#include <stdio.h>
#include<stdlib.h>

struct Node {
    int data;
    struct Node *next;
};
struct Node *head = NULL;

int isEmpty(){
    return head==NULL;
}

void push(int val){
    struct Node *newNode = (struct Node*)malloc(sizeof(struct Node));
    if(newNode == NULL){
        printf("Stack overflow.");
        return;
    }
    newNode->data = val;
    newNode->next = head;
    head = newNode;
    printf("%d pushed onto stack\n",val);
}

void pop(){
    if(isEmpty()){
        printf("Stack underflow.");
        return;
    }
    struct Node *temp = head;
    int val = temp->data;
    head = head->next;
    free(temp);
    printf("Popped element: %d\n",val);
}

void display(){
    if(isEmpty()){
        printf("Stack empty.");
        return;
    }
    struct Node *temp = head;
    printf("Stack elements:\n");
    while(temp != NULL){
        printf("%d\t",temp->data);
        temp = temp->next;
    }
    printf("\n");
}


int main() {
    int val,choice=1;
    while(choice != 4){
        printf("\n--Linked list menu--\n");
        printf("1.Push\n2.Pop\n3.Display\n4.Exit\n");
        printf("Enter your choice: ");
        scanf("%d",&choice);
        switch(choice){
            case 1:
                printf("Enter val to push: ");
                scanf("%d",&val);
                push(val);
                break;
            case 2:
                pop();
                break;
            case 3:
                display();
                break;
            case 4:
                printf("Exiting\n");
                break;
            default: 
                printf("Invalid choice.");
        }
    }
    return 0;
}
